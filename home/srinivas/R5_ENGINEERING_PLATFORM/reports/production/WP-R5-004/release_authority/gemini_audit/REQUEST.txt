def _sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def frame_fingerprint(frames: dict) -> dict:
    """Deterministic content fingerprint of {name: DataFrame} (dtypes + per-row hashes + index)."""
    import pandas as pd
    out = {}
    for name in sorted(frames):
        f = frames[name]
        payload = json.dumps([list(map(str, f.columns)), list(map(str, f.dtypes)), len(f)]).encode()
        out[name] = _sha(payload + pd.util.hash_pandas_object(f, index=True).values.tobytes())
    return out


def _fd_map() -> dict:
    out = {}
    for name in os.listdir("/proc/self/fd"):
        try:

        except OSError:
            pass
    return out


_SELF_FD = re.compile(r"inherited fd (\d+) -> (anon_inode:\[eventpoll\]|pipe:\[\d+\])")


def post_run_witness(pre_fds: dict) -> dict:
    """Final live boundary re-probe. The strict ``active_witness`` runs before the engine imports anything. Importing the
    pinned snapshot's broker stack (tripwire targets, ``kiteconnect`` -> ``twisted.internet.reactor``) legitimately creates ONE
    process-owned epoll fd and ONE waker pipe pair. They are not inherited, name no file/socket/host path and cannot reach
    outside. Only new fds of exactly those two kinds are tolerated; every other probe violation (namespaces, mounts, caps,
    interfaces, root listing, any other fd) still fails."""
    expectation = osi._ACTIVE["expectation"]
    bad = osi.verify_probe(osi.probe_self(), expectation)
    tolerated = [b for b in bad if (m := _SELF_FD.fullmatch(b)) and m.group(1) not in pre_fds]
    rest = [b for b in bad if b not in tolerated]
    if rest:
        raise osi.WitnessError("live OS boundary no longer matches the host observation after the run: " + "; ".join(rest))
    return dict(osi._ACTIVE["witness"], reverified_live=True, post_run_self_created_fds=tolerated,
                post_run_fd_note="process-owned epoll/pipe from the snapshot broker import (twisted reactor); no file or socket")

def parse_manifests(stock_bytes: bytes, grid_bytes: bytes, protocol: dict):
    """Pinned stock/grid metadata manifests -> (sorted stock symbols, sorted grid feed names). Opens no market data."""
    try:
        stock, grid = json.loads(stock_bytes), json.loads(grid_bytes)
        files, mhash = stock["files"], stock["manifest_hash"]
        pinned = protocol["stock_dataset"]["manifest_hash"]
        symbols = [f["symbol"] for f in files]
        entries = grid["files"]
        names = [e["name"] for e in entries]
        sealed = protocol["grid_dataset"]["feeds"]
    except (ValueError, KeyError, TypeError) as exc:
        raise CuratedAuthorizationError(f"manifest metadata malformed: {exc!r}") from None
    if not (h.is_sha256(pinned) and h.is_sha256(mhash)) or mhash != pinned:
        raise CuratedAuthorizationError("stock manifest identity differs from the protocol pin")
    if h.sha256_bytes(json.dumps(files, sort_keys=True, default=str).encode()) != pinned:
        raise CuratedAuthorizationError("stock manifest records do not reproduce the pinned manifest hash")
    if len(set(symbols)) != len(symbols) or stock.get("symbol_count") != len(symbols):
        raise CuratedAuthorizationError("stock manifest symbol set inconsistent")
    if "symbols" in protocol["stock_dataset"] and int(protocol["stock_dataset"]["symbols"]) != len(symbols):
        raise CuratedAuthorizationError("stock manifest symbol count differs from protocol")
    if len(set(names)) != len(names) or set(names) != set(sealed):
        raise CuratedAuthorizationError("grid feed set differs from the sealed protocol")
    for entry in entries:
        if not (h.is_sha256(entry.get("sha256")) and entry["sha256"] == sealed[entry["name"]].get("sha256")):
            raise CuratedAuthorizationError(f"{entry['name']}: grid manifest sha256 differs from the sealed protocol")
    return sorted(symbols), sorted(names)

# --------------------------------------------------------------------------- loaders (EOF artifacts)
def _open_reader(path, art: CuratedArtifact, chunk_size):
    if os.stat(path).st_size != art.size_bytes:
        raise CuratedArtifactError(f"{art.alias}: size differs from the authorization")
    ARTIFACT_OPENS.append(art.alias)
    # cap = size + 1: the extra byte only allows the zero-length read that proves EOF; growth past the size fails closed below
    return h.BoundedLineReader(path, chunk_size=chunk_size, max_physical_bytes=art.size_bytes + 1,
                               expected_commitment=art.sha256)


def _finish_reader(reader, art: CuratedArtifact, data_rows, first_raw, last_raw):
    if reader.rows - 1 != data_rows:
        raise h.ShapeError(f"{art.alias}: {reader.rows - 1} data lines but {data_rows} parsed rows (blank/unparsed lines)")
    if not reader.reached_eof or reader.physical_bytes_read != art.size_bytes or reader.consumed_bytes != art.size_bytes:
        raise CuratedArtifactError(f"{art.alias}: artifact not consumed exactly to EOF")
    ledger_row = reader.record(strict_prefix=False)         # also verifies consumed bytes == full-file sha256 commitment
    if data_rows != art.row_count:
        raise CuratedArtifactError(f"{art.alias}: row count {data_rows} != authorized {art.row_count}")
    if first_raw != art.first_timestamp or last_raw != art.last_timestamp:
        raise CuratedArtifactError(f"{art.alias}: timestamp extent {first_raw!r}..{last_raw!r} != authorized "
                                   f"{art.first_timestamp!r}..{art.last_timestamp!r}")
    return ledger_row


def load_curated_stock(path, art: CuratedArtifact, sessions, warmup_bars, *, train_start, train_end_exclusive,
                       pretrain_policy, chunk_size=h.DEFAULT_CHUNK):
    """Whole artifact, EOF-terminated. Returns (frame, audit, ledger_row). The last row is the engine terminal bar."""
    import pandas as pd
    w = h._worker()
    target_dates = [pd.Timestamp(x).date() for x in sessions]
    target_set = set(target_dates)
    first_date, last_date = target_dates[0], target_dates[-1]
    end_date, start_date = pd.Timestamp(train_end_exclusive).date(), pd.Timestamp(train_start).date()
    if last_date >= end_date:
        raise h.ChronologyError(f"{art.alias}: block extends to/after TRAIN end {end_date}")
    warmup, target, seen, pretrain_rows, previous = deque(maxlen=warmup_bars), [], set(), 0, None
    first_raw = last_raw = None
    started, parsed = False, 0
    reader = _open_reader(path, art, chunk_size)
    try:
        rows = h._dict_reader(reader, art.alias, ["open", "high", "low", "close", "volume"])
        fields = rows.fieldnames
        ts_column = "timestamp" if "timestamp" in fields else "date" if "date" in fields else None
        if ts_column is None:
            raise h.ShapeError(f"{art.alias}: no timestamp/date column")
        for row in rows:
            h._checked(row, art.alias)
            parsed += 1
            ts = w.exchange_ts(row[ts_column])
            first_raw = row[ts_column] if first_raw is None else first_raw
            last_raw = row[ts_column]
            if previous is not None and ts <= previous:
                raise h.ChronologyError(f"{art.alias}: timestamps not strictly increasing at {ts}")
            previous = ts
            day = ts.date()
            if day >= end_date:
                raise h.ChronologyError(f"{art.alias}: row {day} at/after TRAIN end {end_date}")
            if day < start_date:
                if pretrain_policy != "WARMUP_PROVENANCE_NO_FIT":
                    raise h.PreTrainSelectionError(f"{art.alias}: preTRAIN row {day} present but not authorized")
                if day >= first_date:
                    raise h.PreTrainSelectionError(f"{art.alias}: preTRAIN row {day} is not warmup")
                pretrain_rows += 1
            if not started:
                if day < first_date:
                    warmup.append(row)
                    continue
                if day > first_date:
                    raise ValueError(f"{art.alias}: first target session missing {first_date}")
                started = True
            if day > last_date:
                raise CuratedArtifactError(f"{art.alias}: row {day} after the last block session {last_date} "
                                           "(the artifact must end with the block)")
            if day not in target_set:
                raise ValueError(f"{art.alias}: unexpected session {day} inside declared block")
            target.append(row)
            seen.add(day)
        ledger_row = _finish_reader(reader, art, parsed, first_raw, last_raw)
    finally:
        reader.close()
    if pretrain_rows != art.pretrain_rows:
        raise CuratedArtifactError(f"{art.alias}: preTRAIN rows {pretrain_rows} != authorized {art.pretrain_rows}")
    if len(warmup) != warmup_bars:
        raise ValueError(f"{art.alias}: expected {warmup_bars} warmup bars, got {len(warmup)}")
    if seen != target_set:
        raise ValueError(f"{art.alias}: missing target sessions {sorted(target_set - seen)}")
    if len(target) < 2:
        raise ValueError(f"{art.alias}: fewer than two block rows (no trading clock event possible)")
    frame = w.convert_stock_rows(list(warmup) + target)
    warm_frame, target_frame = frame.iloc[:warmup_bars], frame.iloc[warmup_bars:]
    counts = {}
    for ts in target_frame["timestamp"]:
        key = str(w.exchange_ts(ts).date())
        counts[key] = counts.get(key, 0) + 1
    audit = {
        "warmup_rows": len(warm_frame), "warmup_first_utc": w.utc_iso(warm_frame["timestamp"].iloc[0]),
        "warmup_last_utc": w.utc_iso(warm_frame["timestamp"].iloc[-1]), "target_rows": len(target_frame),
        "clock_rows": len(target_frame) - 1, "target_first_utc": w.utc_iso(target_frame["timestamp"].iloc[0]),
        "target_last_utc": w.utc_iso(target_frame["timestamp"].iloc[-1]), "target_session_counts": counts,
        "terminal_bar_policy": TERMINAL_BAR_POLICY, "terminal_sentinel_row_present": False, "ends_at_eof": True,
        "artifact_rows": reader.rows - 1, "artifact_pretrain_rows": pretrain_rows,
        "consumed_prefix_sha256": ledger_row["consumed_prefix_sha256"], "consumed_bytes": ledger_row["consumed_bytes"],
        "physical_bytes_read": ledger_row["physical_bytes_read"], "pretrain_policy": pretrain_policy,
        "warmup_rows_dated_before_train_start": sum(1 for r in warmup if w.exchange_ts(r[ts_column]).date() < start_date),
        "train_scope": h.TRAIN_SCOPE_DISCLOSURE,
    }
    return frame, audit, ledger_row


def load_curated_grid(path, art: CuratedArtifact, name, protocol, cutoff_utc, *, chunk_size=h.DEFAULT_CHUNK):
    """Whole grid artifact validated to EOF; rows available after ``cutoff_utc`` are validated then discarded, never retained."""
    import pandas as pd

def _under(path, roots):
    real = os.path.realpath(path)
    return any(real == r or real.startswith(r.rstrip("/") + "/") for r in roots)


def audit_source_roots(allowed_roots) -> dict:
    """Every sys.path entry and every imported module/package origin must lie under the pinned mounts (code roots + runtime)."""
    roots = tuple(os.path.realpath(r) for r in allowed_roots)
    bad_path = [p for p in sys.path if not _under(p or os.getcwd(), roots)]
    bad_modules = []
    for name, module in list(sys.modules.items()):
        origins = [getattr(module, "__file__", None), *list(getattr(module, "__path__", None) or [])]
        bad_modules += [f"{name}:{o}" for o in origins if o and not _under(o, roots)]
    if bad_path or bad_modules:
        raise SourceRootEscape(f"sources outside the pinned mounts: sys.path={bad_path[:5]} modules={sorted(bad_modules)[:5]}")
    return dict(allowed_roots=list(roots), sys_path=list(sys.path),
                module_origins=sum(1 for m in sys.modules.values() if getattr(m, "__file__", None)))

"""In-sandbox bootstrap (PID 2 inside the bwrap namespaces). Never run outside the launcher.

1. probe the live kernel state and announce READY on stderr (the launcher then observes the sandbox from the host);
2. read ONE handshake line from stdin (stdin is then pointed at /dev/null);
3. re-verify the live boundary against the host observation, the code-tree pins and the input sha256 pins;
4. only then run the target in-process. Any failure => exit 97 with a BOOTSTRAP_FAILED marker; the target never starts.
"""
import json
import os
import runpy
import sys
import traceback

HERE = os.path.dirname(os.path.abspath(__file__))
sys.dont_write_bytecode = True
sys.path.insert(0, HERE)
import os_isolation as osi  # noqa: E402


def fail(reason, **extra):
    sys.stderr.write(f"{osi.MARK_FAIL} {json.dumps(dict(reason=reason, **extra))}@@\n")
    sys.stderr.flush()
    os._exit(osi.BOOTSTRAP_FAIL_RC)


def main():
    probe = osi.probe_self()
    sys.stderr.write(f"{osi.MARK_READY} {json.dumps(dict(ns=probe['ns']))}@@\n")
    sys.stderr.flush()
    raw = sys.stdin.buffer.readline(osi.MAX_HANDSHAKE + 1)
    if not raw.endswith(b"\n") or len(raw) > osi.MAX_HANDSHAKE:
        fail("handshake missing or oversize")
    try:
        hand = json.loads(raw)
        exp, witness = hand["expectation"], hand["witness"]
        nonce, target = hand["nonce"], hand["target"]
        if witness.get("nonce") != nonce or witness.get("schema") != osi.WITNESS_SCHEMA:
            fail("witness nonce/schema mismatch")
        pins, inputs, args = hand["code_pins"], hand["inputs"], hand["target_args"]
    except (ValueError, KeyError, TypeError, AttributeError) as exc:
        fail("malformed handshake", error=str(exc))
    null = os.open("/dev/null", os.O_RDONLY)
    os.dup2(null, 0)
    os.close(null)
    bad = osi.verify_probe(osi.probe_self(), exp)
    if bad:
        fail("live boundary differs from host observation (forged or wrong witness)", violations=bad)
    for name, pin in sorted(pins.items()):
        try:
            actual = osi.tree_sha256(f"{osi.CODE_MOUNT}/{name}")
        except (OSError, osi.OSIsolationError) as exc:
            fail("code root unreadable", root=name, error=str(exc))
        if actual != pin:
            fail("code tree sha256 differs from pin", root=name, expected=pin, actual=actual)
    for alias, pin in sorted(inputs.items()):
        try:
            actual = osi.sha256_file(f"{osi.IN_MOUNT}/{alias}")
        except OSError as exc:
            fail("input unreadable", alias=alias, error=str(exc))
        if actual != pin:
            fail("input sha256 differs from pin", alias=alias, expected=pin, actual=actual)
    if sorted(os.listdir(osi.IN_MOUNT) if os.path.isdir(osi.IN_MOUNT) else []) != sorted(inputs):
        fail("/in content differs from the pinned inputs")
    osi.activate(exp, dict(witness, bootstrap_verified_live=True, verified_code_pins=pins,
                           verified_input_sha256=inputs))
    sys.argv = [target, *args]
    sys.stderr.write(f"{osi.MARK_START}@@\n")
    sys.stderr.flush()
    rc = 0
    try:
        runpy.run_path(target, run_name="__main__")
    except SystemExit as exc:
        rc = exc.code if isinstance(exc.code, int) else (0 if exc.code is None else 1)
    except BaseException:
        traceback.print_exc()
        rc = 1
    sys.stdout.flush()
    sys.stderr.write(f"{osi.MARK_END} {json.dumps(dict(rc=rc))}@@\n")
    sys.stderr.flush()
    os._exit(rc & 0xFF)


if __name__ == "__main__":
    main()

def read_fixed_store(forbidden_roots=(), forbidden_files=()) -> bytes:
    """HOST: bytes of the store at the fixed path. Refuses a missing/empty/non-plain/oversized/group-or-world-writable file and
    any store that lives inside caller-reachable trees (the pinned code roots, the output dir) or IS a caller-supplied file."""

    try:
        st = os.lstat(path)
    except OSError:
        raise ApprovalError("approval store missing") from None
    if not stat.S_ISREG(st.st_mode):
        raise ApprovalError("approval store is not a plain file (symlink/directory refused)")
    if st.st_mode & 0o022:
        raise ApprovalError("approval store is group/world writable")
    if st.st_size == 0:
        raise ApprovalError("approval store is empty")
    if st.st_size > MAX_STORE_BYTES:
        raise ApprovalError("approval store too large")
    real = os.path.realpath(path)
    for root in forbidden_roots:
        root = os.path.realpath(root)
        if real == root or real.startswith(root.rstrip("/") + "/"):
            raise ApprovalError(f"approval store lies inside a caller-reachable tree ({root}): self-pinned authority refused")

        raise ApprovalError("approval store is a caller-supplied file: self-pinned authority refused")
    with open(path, "rb") as fh:
        data = fh.read(MAX_STORE_BYTES + 1)
    if len(data) != st.st_size:
        raise ApprovalError("approval store changed while reading")
    return data

def _positive_int(value, label):

    if isinstance(value, bool) or not isinstance(value, int) or value < 1:

def normalize_intervals(intervals):
    """Sorted, non-overlapping, integral half-open [start, end) byte intervals tagged with a kind."""
    out, last_end = [], 0
    for item in intervals:
        try:
            start, end, kind = item
        except (TypeError, ValueError):
            raise ReadBoundaryError(f"malformed interval {item!r}") from None
        for value, label in ((start, "start"), (end, "end")):
            if isinstance(value, bool) or not isinstance(value, int):
                raise ReadBoundaryError(f"interval {label} must be an integer, got {value!r}")
        if kind not in INTERVAL_KINDS:
            raise ReadBoundaryError(f"unknown interval kind {kind!r}")
        if start < 0 or end <= start or start < last_end:
            raise ReadBoundaryError(f"intervals must be positive-length, sorted, non-overlapping: {item!r}")
        out.append((start, end, kind))
        last_end = end
    if not out:
        raise ReadBoundaryError("empty interval list")
    return tuple(out)

def verify_snapshot(snapshot_dir, manifest_path=None, *, expected_manifest_sha256=None) -> dict:
    """Re-hash every snapshot file against SNAPSHOT_MANIFEST.json; fail closed."""

class BoundedLineReader:
    """Binary line reader over an unbuffered file: counts every physical byte and hashes consumed lines."""

    def __init__(self, path, *, chunk_size=DEFAULT_CHUNK, max_physical_bytes=None, intervals=None,
                 expected_commitment=None):
        # Every argument is validated BEFORE the file is opened or stat'ed.
        self.chunk_size = _positive_int(chunk_size, "chunk_size")
        self.max_physical_bytes = None if max_physical_bytes is None else _positive_int(max_physical_bytes, "max_physical_bytes")
        self.intervals = None if intervals is None else normalize_intervals(intervals)
        if expected_commitment is not None and not is_sha256(expected_commitment):
            raise HarnessError("malformed expected content commitment")
        self.expected_commitment = expected_commitment
        self.path = Path(path)
        self.size_bytes = self.path.stat().st_size
        if self.intervals is not None and self.intervals[-1][1] > self.size_bytes:
            raise ReadBoundaryError(f"{self.path.name}: authorized interval ends beyond file size")
        self.physical_bytes_read = 0
        self.read_calls = 0
        self.consumed_bytes = 0
        self.rows = 0
        self.reached_eof = False
        self.line_kind = None          # kind of the authorized interval the most recently consumed line came from
        self.first_data_offset = None
        self.read_ranges = []          # (offset, length) of every physical read, in order
        self._buffer = b""
        self._digest = hashlib.sha256()
        self._ivl, self._pos, self._kind, self._read_kind = -1, 0, None, None
        _IN_HARNESS_READ.active = True
        try:
            self._handle = builtins.open(self.path, "rb", buffering=0)
        finally:
            _IN_HARNESS_READ.active = False

    def _next_read_size(self) -> int:
        """Bytes the next physical read may request: min(chunk, interval remaining, cap remaining); fails closed at 0."""
        cap_left = None if self.max_physical_bytes is None else self.max_physical_bytes - self.physical_bytes_read
        plan_left = None
        if self.intervals is not None:
            count = len(self.intervals)
            while self._ivl < 0 or (self._ivl < count and self._pos >= self.intervals[self._ivl][1]):
                nxt = self._ivl + 1
                if nxt >= count:
                    self._ivl = count
                    break
                if self._buffer:
                    raise ReadBoundaryError(f"{self.path.name}: line straddles an authorized interval boundary")
                start, _end, kind = self.intervals[nxt]
                if self._pos != start:
                    self._handle.seek(start)       # repositioning only; no bytes are read
                    self._pos = start
                self._ivl, self._kind = nxt, kind
            plan_left = 0 if self._ivl >= count else self.intervals[self._ivl][1] - self._pos
        limit = self.chunk_size
        for bound in (plan_left, cap_left):
            if bound is not None:
                limit = min(limit, bound)
        if limit <= 0:
            if plan_left is not None and plan_left <= 0:
                raise ReadBoundaryError(f"{self.path.name}: authorized read plan exhausted")
            raise ReadBoundaryError(f"{self.path.name}: physical read cap {self.max_physical_bytes} reached")
        return limit

    def next_line(self):
        while True:
            index = self._buffer.find(b"\n")
            if index >= 0:
                line, self._buffer = self._buffer[:index + 1], self._buffer[index + 1:]
                return self._consume(line)
            if len(self._buffer) > MAX_LINE_BYTES:
                raise ReadBoundaryError(f"{self.path.name}: line exceeds {MAX_LINE_BYTES} bytes")
            if self.reached_eof:
                if self._buffer:
                    line, self._buffer = self._buffer, b""
                    return self._consume(line)
                return None
            limit = self._next_read_size()       # raises BEFORE the first prohibited byte is requested
            chunk = self._handle.read(limit)
            self.read_calls += 1
            if len(chunk) > limit:
                raise ReadBoundaryError(f"{self.path.name}: read returned {len(chunk)} > requested {limit}")
            if not chunk:
                if self.intervals is not None:
                    raise ReadBoundaryError(f"{self.path.name}: file ended inside an authorized interval")
                self.reached_eof = True
            else:
                self.read_ranges.append((self._pos, len(chunk)))
                self._pos += len(chunk)
                self._read_kind = self._kind
            self.physical_bytes_read += len(chunk)
            if self.max_physical_bytes is not None and self.physical_bytes_read > self.max_physical_bytes:
                raise ReadBoundaryError(f"{self.path.name}: physical read cap {self.max_physical_bytes} exceeded")
            self._buffer += chunk

    def _consume(self, line: bytes) -> bytes:
        if len(line) > MAX_LINE_BYTES:
            raise ReadBoundaryError(f"{self.path.name}: line exceeds {MAX_LINE_BYTES} bytes")
        if self.first_data_offset is None and self._read_kind != "HEADER":
            # lines never straddle intervals, so the first non-header line starts its interval
            self.first_data_offset = self.intervals[self._ivl][0] if self.intervals is not None else 0
        self.line_kind = self._read_kind
        self._digest.update(line)
        self.consumed_bytes += len(line)
        self.rows += 1
        return line

    @property
    def data_contiguous_from_start(self) -> bool:
        """True when the first non-header row sits directly after the header (nothing skipped from file start)."""
        if self.intervals is None:
            return True
        header_end = self.intervals[0][1] if self.intervals[0][2] == "HEADER" else 0
        return self.first_data_offset is not None and self.first_data_offset == header_end

    def close(self):
        self._handle.close()

    def record(self, *, strict_prefix=True) -> dict:
        overread = self.physical_bytes_read - self.consumed_bytes
        if overread < 0 or overread >= self.chunk_size:
            raise ReadBoundaryError(f"{self.path.name}: overread {overread} outside [0,{self.chunk_size})")
        if self.max_physical_bytes is not None and self.physical_bytes_read > self.max_physical_bytes:
            raise ReadBoundaryError(f"{self.path.name}: physical read cap exceeded")
        if strict_prefix and (self.reached_eof or self.physical_bytes_read >= self.size_bytes):
            raise ReadBoundaryError(f"{self.path.name}: bounded access not proven (read reached end of file)")
        digest = self._digest.hexdigest()
        planned = None
        if self.intervals is not None:
            planned = sum(end - start for start, end, _ in self.intervals)
            if self.physical_bytes_read != planned or self.consumed_bytes != planned:
                raise ReadBoundaryError(
                    f"{self.path.name}: authorized plan not exactly consumed (planned={planned} "
                    f"physical={self.physical_bytes_read} consumed={self.consumed_bytes})")
        if self.expected_commitment is not None and digest != self.expected_commitment:
            raise ContentCommitmentError(f"{self.path.name}: consumed bytes do not match supplied content commitment")
        return dict(path=str(self.path), size_bytes=self.size_bytes, chunk_size=self.chunk_size,
                    physical_bytes_read=self.physical_bytes_read, consumed_bytes=self.consumed_bytes,
                    overread_bytes=overread, read_calls=self.read_calls, lines_consumed=self.rows,
                    reached_eof=self.reached_eof, consumed_prefix_sha256=digest,
                    max_physical_bytes=self.max_physical_bytes, planned_bytes=planned,
                    read_ranges=[list(r) for r in self.read_ranges],
                    content_commitment_verified=self.expected_commitment is not None)


def _csv_lines(reader: BoundedLineReader, name: str):
    """Decoded, shape-checked text lines for csv; stops exactly where the consumer stops."""
    first = True
    while True:
        raw = reader.next_line()
        if raw is None:
            return
        body = raw[3:] if first and raw.startswith(b"\xef\xbb\xbf") else raw
        first = False
        text = body.decode("utf-8")
        text = text[:-2] if text.endswith("\r\n") else text[:-1] if text.endswith("\n") else text
        if "\r" in text or '"' in text:
            raise ShapeError(f"{name}: unsupported quoting/line terminator in row {reader.rows}")
        yield text


def _dict_reader(reader, name, required):
    dict_reader = csv.DictReader(_csv_lines(reader, name))
    fields = dict_reader.fieldnames or []
    if len(set(fields)) != len(fields):
        raise ShapeError(f"{name}: duplicate header columns")
    missing = [c for c in required if c not in fields]
    if missing:
        raise ShapeError(f"{name}: missing columns {missing}")
    return dict_reader


def _checked(row, name):
    if None in row or any(v is None for v in row.values()):
        raise ShapeError(f"{name}: row shape differs from header")
    return row

def _reader_for(path, file_plan, chunk_size, max_physical_bytes):
    """Reader bound to the per-file authorization when one exists (plan lookup precedes any open)."""
    if file_plan is None:
        return BoundedLineReader(path, chunk_size=chunk_size, max_physical_bytes=max_physical_bytes)

                             intervals=file_plan.intervals, expected_commitment=file_plan.content_commitment_sha256)


def load_stock_block_bounded(path, sessions, warmup_bars, *, train_end_exclusive, record=None,
                             chunk_size=DEFAULT_CHUNK, max_physical_bytes=None, strict_prefix=True,
                             file_plan=None, train_start=None, pretrain_policy="REJECT"):
    """Chronological prefix of one stock file: warmup + target sessions + exactly one sentinel."""
    import pandas as pd
    w = _worker()
    path = Path(path)
    target_dates = [pd.Timestamp(x).date() for x in sessions]
    target_set = set(target_dates)
    first_date, last_date = target_dates[0], target_dates[-1]
    end_date = pd.Timestamp(train_end_exclusive).date()
    if last_date >= end_date:
        raise ChronologyError(f"{path.name}: block extends to/after TRAIN end {end_date}")
    from collections import deque
    warmup = deque(maxlen=warmup_bars)
    target, sentinel, seen = [], None, set()
    pretrain_start = pd.Timestamp(train_start).date() if train_start is not None else None
    pretrain_rows = 0
    reader = _reader_for(path, file_plan, chunk_size, max_physical_bytes)
    try:
        rows = _dict_reader(reader, path.name, ["open", "high", "low", "close", "volume"])
        fieldnames = rows.fieldnames
        ts_column = "timestamp" if "timestamp" in fieldnames else "date" if "date" in fieldnames else None
        if ts_column is None:
            raise ShapeError(f"{path}: no timestamp/date column")
        started, previous, first_raw = False, None, None
        for row in rows:
            _checked(row, path.name)
            ts = w.exchange_ts(row[ts_column])
            if first_raw is None:
                first_raw = row[ts_column]
                # only meaningful when nothing was skipped between the header and the first data row
                if record is not None and reader.data_contiguous_from_start and first_raw != record.first_timestamp:
                    raise ManifestMismatchError(f"{path.name}: first row {first_raw!r} != manifest {record.first_timestamp!r}")
            if previous is not None and ts <= previous:
                raise ChronologyError(f"{path.name}: timestamps not strictly increasing at {ts}")
            previous = ts
            day = ts.date()
            if day >= end_date:
                raise ChronologyError(f"{path.name}: row {day} at/after TRAIN end {end_date}")
            if pretrain_start is not None:
                kind = reader.line_kind
                if day < pretrain_start:
                    if pretrain_policy != "WARMUP_PROVENANCE_NO_FIT":
                        raise PreTrainSelectionError(f"{path.name}: preTRAIN row {day} selected but not authorized")
                    if reader.intervals is not None and kind != "PRETRAIN_WARMUP_NO_FIT":
                        raise PreTrainSelectionError(f"{path.name}: preTRAIN row {day} outside a PRETRAIN_WARMUP_NO_FIT interval")
                    if day >= first_date:
                        raise PreTrainSelectionError(f"{path.name}: preTRAIN row {day} is not warmup")
                    pretrain_rows += 1
                elif kind == "PRETRAIN_WARMUP_NO_FIT":
                    raise PreTrainSelectionError(f"{path.name}: TRAIN-dated row {day} inside a PRETRAIN interval")
            if not started:
                if day < first_date:
                    warmup.append(row)
                    continue
                if day > first_date:
                    raise ValueError(f"{path.name}: first target session missing {first_date}")
                started = True
            if day <= last_date:
                if day not in target_set:
                    raise ValueError(f"{path.name}: unexpected real session {day} inside declared block")
                target.append(row)
                seen.add(day)
                continue
            sentinel = row
            break
        ledger_row = reader.record(strict_prefix=strict_prefix) if sentinel is not None else None
    finally:
        reader.close()
    if len(warmup) != warmup_bars:
        raise ValueError(f"{path.name}: expected {warmup_bars} warmup bars, got {len(warmup)}")
    if seen != target_set:
        raise ValueError(f"{path.name}: missing target sessions {sorted(target_set - seen)}")
    if not target:
        raise ValueError(f"{path.name}: zero target rows")
    if sentinel is None:
        raise ValueError(f"{path.name}: boundary sentinel missing")
    if record is not None:
        last_allowed = w.exchange_ts(record.last_timestamp)
        if w.exchange_ts(sentinel[ts_column]) > last_allowed or reader.rows > record.row_count:
            raise ManifestMismatchError(f"{path.name}: consumed rows exceed manifest extent")
    warmup_pretrain = (sum(1 for r in warmup if w.exchange_ts(r[ts_column]).date() < pretrain_start)
                       if pretrain_start is not None else None)
    frame = w.convert_stock_rows(list(warmup) + target + [sentinel])
    warm_frame, target_frame, sentinel_frame = frame.iloc[:warmup_bars], frame.iloc[warmup_bars:-1], frame.iloc[-1:]
    counts = {}
    for ts in target_frame["timestamp"]:
        key = str(w.exchange_ts(ts).date())
        counts[key] = counts.get(key, 0) + 1
    if w.exchange_ts(sentinel_frame["timestamp"].iloc[0]) <= w.exchange_ts(target_frame["timestamp"].iloc[-1]):
        raise ValueError(f"{path.name}: sentinel is not after target")
    audit = {
        "warmup_rows": len(warm_frame), "warmup_first_utc": w.utc_iso(warm_frame["timestamp"].iloc[0]),
        "warmup_last_utc": w.utc_iso(warm_frame["timestamp"].iloc[-1]), "target_rows": len(target_frame),
        "target_first_utc": w.utc_iso(target_frame["timestamp"].iloc[0]),
        "target_last_utc": w.utc_iso(target_frame["timestamp"].iloc[-1]), "target_session_counts": counts,
        "sentinel_rows": 1, "sentinel_utc": w.utc_iso(sentinel_frame["timestamp"].iloc[0]),
        "consumed_prefix_sha256": ledger_row["consumed_prefix_sha256"], "consumed_bytes": ledger_row["consumed_bytes"],
        "physical_bytes_read": ledger_row["physical_bytes_read"],
        "pretrain_policy": pretrain_policy, "pretrain_rows_consumed": pretrain_rows,
        "warmup_rows_dated_before_train_start": warmup_pretrain,
        "sentinel_row_authorized": None if file_plan is None else file_plan.terminal_boundary_row_authorized,
        "train_scope": TRAIN_SCOPE_DISCLOSURE,
    }
    return frame, audit, ledger_row


def load_grid_prefix_bounded(records, protocol, cutoff_utc, *, allowed_grid_roots, chunk_size=DEFAULT_CHUNK,
                             max_physical_bytes=None, strict_prefix=True, read_plan=None, train_start=None):
    """Causal Nifty/VIX prefix. Declared sha256 values are compared as METADATA only (never re-hashed)."""
    import pandas as pd
    expected = protocol["grid_dataset"]["feeds"]
    delay = int(protocol["grid_dataset"]["availability_delay_minutes"])
    _, end_date = _train_bounds(protocol)
    grid_train_start = pd.Timestamp(train_start).date() if train_start is not None else None
    feeds, audit, ledger_rows = {}, {}, []
    for name in sorted(expected):
        record, sealed = records[name], expected[name]
        if not is_sha256(sealed.get("sha256")) or not is_sha256(record.get("sha256")):
            raise ManifestMismatchError(f"{name}: malformed grid sha256")
        if record["sha256"] != sealed["sha256"]:
            raise ValueError(f"{name}: local manifest SHA mismatch")
        path = _safe_file(Path(record["path"]), allowed_grid_roots, f"grid feed {name}")
        ts_column = sealed["timestamp_column"]
        file_plan = read_plan.require(path) if read_plan is not None else None
        reader = _reader_for(path, file_plan, chunk_size, max_physical_bytes)
        rows, previous, boundary_rows = [], None, 0
        try:
            dict_reader = _dict_reader(reader, name, [ts_column, "close"])
            for row in dict_reader:
                _checked(row, name)
                source = pd.to_datetime(row[ts_column], utc=True)
                if source.tz_convert("Asia/Kolkata").date() >= end_date:
                    raise ChronologyError(f"{name}: grid row {source} at/after TRAIN end {end_date}")
                if grid_train_start is not None and (source.tz_convert("Asia/Kolkata").date() < grid_train_start
                                                     or reader.line_kind == "PRETRAIN_WARMUP_NO_FIT"):
                    raise PreTrainSelectionError(f"{name}: preTRAIN grid row {source} would enter a retained feed")
                availability = source + pd.Timedelta(minutes=delay)
                if previous is not None and availability <= previous:
                    raise ChronologyError(f"{name}: grid timestamps not strictly increasing at {availability}")
                previous = availability
                if availability > cutoff_utc:
                    boundary_rows = 1  # one row is read to detect the cutoff; it is discarded, never retained
                    break
                rows.append({"timestamp": availability, "close": row["close"]})
            ledger_row = reader.record(strict_prefix=strict_prefix)
        finally:
            reader.close()
        if not rows:
            raise ValueError(f"{name}: no causal grid prefix")
        if not boundary_rows:
            raise ReadBoundaryError(f"{name}: grid cutoff boundary row not observed")
        frame = pd.DataFrame(rows)
        frame["timestamp"] = pd.to_datetime(frame["timestamp"], utc=True)
        frame["close"] = pd.to_numeric(frame["close"], errors="raise")
        if (frame["timestamp"] > cutoff_utc).any():
            raise ValueError(f"{name}: post-cutoff row retained")
        feeds[name] = frame
        audit[name] = {
            "declared_sha256_unverified_metadata": sealed["sha256"],
            "consumed_prefix_sha256": ledger_row["consumed_prefix_sha256"], "consumed_bytes": ledger_row["consumed_bytes"],
            "physical_bytes_read": ledger_row["physical_bytes_read"], "rows": len(frame),
            "first_available_utc": frame["timestamp"].iloc[0].isoformat(),
            "last_available_utc": frame["timestamp"].iloc[-1].isoformat(), "cutoff_utc": cutoff_utc.isoformat(),
            "rows_after_cutoff_retained": 0, "boundary_rows_parsed_and_discarded": boundary_rows,
            "boundary_row_authorized": None if file_plan is None else file_plan.terminal_boundary_row_authorized,
            "boundary_row_affects_retained_feed": False, "train_scope": TRAIN_SCOPE_DISCLOSURE,
        }
        ledger_rows.append(ledger_row)
    return feeds, audit, ledger_rows


# ------------------------------------------------------------ roots / manifest
def _inside(path: Path, roots) -> bool:
    return any(path == r or path.is_relative_to(r) for r in roots)


def _resolve_roots(roots, label):
    out = []
    for r in roots:
        r = Path(r)
        if not r.is_absolute() or ".." in r.parts:
            raise UnsafeRootError(f"{label} root must be absolute without '..': {r}")
        out.append(r.resolve())
    if not out:
        raise UnsafeRootError(f"no {label} roots declared")
    return tuple(out)


def _safe_file(path: Path, roots, label) -> Path:
    if not path.is_absolute() or ".." in path.parts:
        raise UnsafeRootError(f"{label}: unsafe path {path}")
    resolved = path.resolve()
    if not _inside(resolved, roots):
        raise UnsafeRootError(f"{label}: {resolved} outside declared roots")
    if resolved != Path(os.path.abspath(path)) or not resolved.is_file():
        raise UnsafeRootError(f"{label}: {path} is not a plain file within roots (symlink/missing)")
    return resolved

class UnplannedFileError(ReadBoundaryError):
    pass


@dataclass(frozen=True)
class FilePlan:
    role: str
    name: str
    path: str
    intervals: tuple
    max_physical_bytes: int
    content_commitment_sha256: str
    provenance_id: str
    boundary_evidence: dict
    terminal_boundary_row_authorized: bool

    @property
    def planned_bytes(self) -> int:
        return sum(end - start for start, end, _ in self.intervals)


@dataclass(frozen=True)
class ReadPlan:
    authorization_sha256: str
    execution_package_id: str
    pretrain_policy: str
    files: dict   # canonical path -> FilePlan

    def require(self, path) -> FilePlan:
        entry = self.files.get(str(path))
        if entry is None:
            raise UnplannedFileError(f"{path}: not in the authorized read plan")
        return entry

    def check_coverage(self, actual: dict):
        """actual: canonical path -> (role, name, size_bytes). Exact bijection with the plan; sizes bound the intervals."""
        if set(actual) != set(self.files):
            raise AuthorizationError(
                f"read plan files differ from metadata plan: missing={sorted(set(actual) - set(self.files))[:5]} "
                f"unexpected={sorted(set(self.files) - set(actual))[:5]}")
        for path, (role, name, size) in actual.items():
            entry = self.files[path]
            if (entry.role, entry.name) != (role, name):
                raise AuthorizationError(f"{path}: plan role/name {(entry.role, entry.name)} != {(role, name)}")
            if entry.intervals[-1][1] >= size:
                raise AuthorizationError(f"{path}: authorized interval does not end strictly before file size {size}")
            if entry.boundary_evidence["file_size_bytes_at_audit"] != size:
                raise AuthorizationError(f"{path}: file size differs from boundary evidence size")

    def receipt(self) -> dict:
        return dict(
            authorization_sha256=self.authorization_sha256, execution_package_id=self.execution_package_id,
            pretrain_policy=self.pretrain_policy, commitment_scope=COMMITMENT_SCOPE,
            commitment_disclosure="a freshly supplied prefix commitment is the CURRENT ACQUISITION identity of the consumed "
                                  "bytes; it is not historical sealed membership and is not independent validation",
            train_scope=TRAIN_SCOPE_DISCLOSURE,
            files=[dict(role=f.role, name=f.name, path=f.path, intervals=[list(i) for i in f.intervals],
                        max_physical_bytes=f.max_physical_bytes, provenance_id=f.provenance_id,
                        content_commitment_sha256=f.content_commitment_sha256,
                        boundary_evidence=dict(f.boundary_evidence),
                        terminal_boundary_row_authorized=f.terminal_boundary_row_authorized)
                   for _, f in sorted(self.files.items())])


def _exact_keys(obj, keys, label):
    if not isinstance(obj, dict) or set(obj) != set(keys):
        extra = sorted(set(obj) - set(keys)) if isinstance(obj, dict) else "not an object"
        missing = sorted(set(keys) - set(obj)) if isinstance(obj, dict) else "-"
        raise AuthorizationError(f"{label}: keys must be exactly {sorted(keys)} (extra={extra} missing={missing})")
    return obj


def _canon_dir(value, label):
    if not isinstance(value, str) or not Path(value).is_absolute() or ".." in Path(value).parts:
        raise AuthorizationError(f"{label}: must be an absolute path without '..'")
    return str(Path(value).resolve())


def _parse_file_entry(entry, pretrain_policy, roots_by_role, feed_names):
    _exact_keys(entry, _FILE_KEYS, "file entry")
    role, name, path = entry["role"], entry["name"], entry["canonical_path"]
    if role not in ("stock", "grid") or not isinstance(name, str) or not name:
        raise AuthorizationError(f"file entry: bad role/name {role!r}/{name!r}")
    if role == "grid" and name not in feed_names:
        raise AuthorizationError(f"grid feed {name!r} not in protocol")
    if not isinstance(path, str) or not Path(path).is_absolute() or ".." in Path(path).parts or str(Path(path).resolve()) != path:
        raise AuthorizationError(f"{name}: canonical_path must be an absolute, already-resolved path")
    if not Path(path).is_relative_to(roots_by_role[role]):
        raise AuthorizationError(f"{name}: canonical_path outside the declared {role} root")
    raw = entry["intervals"]
    if not isinstance(raw, list) or not raw:
        raise AuthorizationError(f"{name}: intervals must be a non-empty list")
    parsed = []
    for item in raw:
        _exact_keys(item, {"start", "end", "kind"}, f"{name}: interval")
        parsed.append((item["start"], item["end"], item["kind"]))
    try:
        intervals = normalize_intervals(parsed)
    except ReadBoundaryError as exc:
        raise AuthorizationError(f"{name}: {exc}") from None
    kinds = [k for _, _, k in intervals]
    if intervals[0][0] != 0 or kinds[0] != "HEADER" or kinds.count("HEADER") != 1:
        raise AuthorizationError(f"{name}: first (and only) HEADER interval must start at byte 0")
    if "PRETRAIN_WARMUP_NO_FIT" in kinds:
        if pretrain_policy != "WARMUP_PROVENANCE_NO_FIT":
            raise AuthorizationError(f"{name}: preTRAIN interval present but pretrain_policy is {pretrain_policy}")
        if role != "stock":
            raise AuthorizationError(f"{name}: preTRAIN rows are never authorized for grid feeds")
    if "TRAIN_DATA" not in kinds:
        raise AuthorizationError(f"{name}: no TRAIN_DATA interval")
    if "PRETRAIN_WARMUP_NO_FIT" in kinds \
            and kinds.index("TRAIN_DATA") < len(kinds) - 1 - kinds[::-1].index("PRETRAIN_WARMUP_NO_FIT"):
        raise AuthorizationError(f"{name}: preTRAIN intervals must precede TRAIN_DATA intervals")
    try:
        limit = _positive_int(entry["max_physical_bytes"], f"{name}: max_physical_bytes")
    except HarnessError as exc:
        raise AuthorizationError(str(exc)) from None
    if limit != sum(end - start for start, end, _ in intervals):
        raise AuthorizationError(f"{name}: max_physical_bytes must equal the total authorized interval bytes")
    if not is_sha256(entry["content_commitment_sha256"]):
        raise AuthorizationError(f"{name}: malformed content_commitment_sha256")
    if not isinstance(entry["provenance_id"], str) or not entry["provenance_id"].strip():
        raise AuthorizationError(f"{name}: provenance_id missing")
    evidence = _exact_keys(entry["boundary_evidence"], _EVIDENCE_KEYS, f"{name}: boundary_evidence")
    if (evidence["schema"] != BOUNDARY_EVIDENCE_SCHEMA or evidence["independent_of_harness"] is not True
            or not is_sha256(evidence["evidence_artifact_sha256"])
            or not all(isinstance(evidence[k], str) and evidence[k].strip() for k in ("evidence_id", "issuer"))):
        raise AuthorizationError(f"{name}: boundary evidence incomplete/invalid (required schema: "
                                 f"{json.dumps(REQUIRED_EXTERNAL_BOUNDARY_EVIDENCE_SCHEMA)})")
    try:
        _positive_int(evidence["file_size_bytes_at_audit"], f"{name}: file_size_bytes_at_audit")
    except HarnessError as exc:
        raise AuthorizationError(str(exc)) from None
    if entry["terminal_boundary_row_authorized"] is not True:
        raise AuthorizationError(f"{name}: terminal boundary row (sentinel / cutoff row) must be explicitly authorized")
    return FilePlan(role=role, name=name, path=path, intervals=intervals, max_physical_bytes=limit,
                    content_commitment_sha256=entry["content_commitment_sha256"], provenance_id=entry["provenance_id"],
                    boundary_evidence=dict(evidence), terminal_boundary_row_authorized=True)

    protocol_sha256, params_sha256, manifest_root, data_root, grid_root, output_root, output_dir.

        raise AuthorizationError("malformed authorization sha256 pin")

    if actual_sha != authorization_sha256_pin:
        raise AuthorizationError("authorization file sha256 differs from pin")

def validate_block(protocol, block, config):

class Tripwire:
    """Counting guard: any CNC/GTT/tick/conversion/network/raw-market-file access raises and is recorded."""

    def __init__(self, forbidden_read_roots=(), allowed_write_roots=(), extra_targets=(), planned_paths=None):
        self.planned_paths = None if planned_paths is None else frozenset(str(p) for p in planned_paths)
        self.counts = {}
        self.violations = []
        self.forbidden_read_roots = tuple(Path(p).resolve() for p in forbidden_read_roots)
        self.allowed_write_roots = tuple(Path(p).resolve() for p in allowed_write_roots)
        self.targets = TRIPWIRE_TARGETS + tuple(extra_targets)
        self._restore = []

    def _trip(self, name):
        self.counts[name] = self.counts.get(name, 0) + 1
        self.violations.append(name)
        raise TripwireViolation(f"forbidden access: {name}")

    def _patch(self, owner, attr, label):
        original = owner.__dict__.get(attr, _MISSING) if isinstance(owner, type) else getattr(owner, attr)
        self._restore.append((owner, attr, original))

        def tripped(*args, **kwargs):
            self._trip(label)
        setattr(owner, attr, tripped)

    def __enter__(self):
        for module_name, dotted in self.targets:
            module = importlib.import_module(module_name)
            parts = dotted.split(".")
            owner = module if len(parts) == 1 else getattr(module, parts[0])
            self._patch(owner, parts[-1], f"{module_name}:{dotted}")
        for owner, attr in ((socket.socket, "connect"), (socket.socket, "connect_ex"),
                            (socket, "create_connection"), (socket, "getaddrinfo")):
            self._patch(owner, attr, f"socket.{attr}")
        original_open = builtins.open
        self._restore.append((builtins, "open", original_open))
        guard = self

        def guarded_open(file, mode="r", *args, **kwargs):
            if (isinstance(file, (str, bytes, os.PathLike)) and getattr(_IN_HARNESS_READ, "active", False)
                    and guard.planned_paths is not None
                    and str(Path(os.fsdecode(file)).resolve()) not in guard.planned_paths):
                guard._trip(f"unplanned_harness_open:{file}")
            if isinstance(file, (str, bytes, os.PathLike)) and not getattr(_IN_HARNESS_READ, "active", False):
                resolved = Path(os.fsdecode(file)).resolve()
                if any(resolved == r or resolved.is_relative_to(r) for r in guard.forbidden_read_roots):
                    guard._trip(f"raw_open:{resolved}")
                if _WRITE_MODES & set(mode) and not any(resolved.is_relative_to(r) for r in guard.allowed_write_roots):
                    guard._trip(f"write_outside_output:{resolved}")
            return original_open(file, mode, *args, **kwargs)
        self._restore.append((io, "open", original_open))
        builtins.open = guarded_open
        io.open = guarded_open
        return self

    def __exit__(self, *exc):
        for owner, attr, original in reversed(self._restore):
            if original is _MISSING:
                delattr(owner, attr)
            else:
                setattr(owner, attr, original)
        self._restore.clear()
        return False

    def report(self):
        return dict(counts=dict(self.counts), violations=list(self.violations), targets=[f"{m}:{d}" for m, d in self.targets],
                    zero_access=not self.violations,
                    kiteconnect_module_loaded_import_only="kiteconnect" in sys.modules)

MAX_HANDSHAKE = 1 << 20

# Directories that must never be bind sources (nor ancestors of protected paths): "broad" mounts.
BROAD_DENY = tuple(sorted({"/", "/home", "/root", "/etc", "/var", "/usr", "/usr/lib", "/usr/bin", "/lib", "/lib64", "/bin",
                           "/sbin", "/opt", "/mnt", "/media", "/srv", "/tmp", "/run", "/boot", "/sys", "/proc", "/dev",
                           "/home/srinivas", str(Path.home())}))
RELEASE_REQUIREMENTS = (
    "independently issued, hash-pinned TRAIN-only standalone input artifacts (stock + grid) whose bytes are exactly the "
    "authorized TRAIN/HEADER/warm-up intervals, with provenance distinct from the original raw files",
    "Director release acceptance record (hash-pinned in os_isolation.REAL_RUN_RELEASE_ACCEPTANCE_SHA256) naming this "
    "isolation profile, the code-tree pins, the input pins and the execution package",
    "an authorization whose roots/canonical_path/interval fields describe the curated artifacts as mounted in the sandbox",
    "a passing OS isolation test receipt for the exact launcher/bootstrap/harness hashes being released",
)
REAL_RUN_RELEASE_ACCEPTANCE_SHA256 = None   # None => real-data release closed. No code path opens it today.


class OSIsolationError(Exception):
    pass


class UnsupportedEnvironment(OSIsolationError):
    pass


class BroadMountError(OSIsolationError):
    pass


class PinMismatch(OSIsolationError):
    pass


class WitnessError(OSIsolationError):
    pass


# --------------------------------------------------------------------- hashing
def sha256_file(path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def tree_sha256(root) -> str:
    """sha256 over sorted (relative path, file sha256) of every regular file; symlinks/specials refuse; __pycache__ ignored."""
    root, rows = Path(root), []
    for dp, dn, fn in os.walk(root, followlinks=False):
        dn[:] = sorted(d for d in dn if d != "__pycache__")
        for name in dn:
            if os.path.islink(os.path.join(dp, name)):
                raise PinMismatch(f"symlinked directory inside code root: {os.path.join(dp, name)}")
        for name in sorted(fn):
            p = os.path.join(dp, name)
            if not stat.S_ISREG(os.lstat(p).st_mode):
                raise PinMismatch(f"non-regular file inside code root: {p}")
            rows.append(f"{os.path.relpath(p, root)}\0{sha256_file(p)}\n")
    return hashlib.sha256("".join(rows).encode()).hexdigest()


# --------------------------------------------------------------------- environment
def check_environment() -> dict:
    """Fail closed (UnsupportedEnvironment) unless the launcher prerequisites are present. Does not launch anything."""
    if not sys.platform.startswith("linux"):
        raise UnsupportedEnvironment("linux required")
    for tool in (BWRAP, LDD):
        if not os.path.isfile(tool) or not os.access(tool, os.X_OK):
            raise UnsupportedEnvironment(f"{tool} missing or not executable")
    if os.stat(BWRAP).st_mode & stat.S_ISUID:
        raise UnsupportedEnvironment("setuid bwrap not supported: the unprivileged user-namespace path is required")
    out = subprocess.run([BWRAP, "--version"], capture_output=True, text=True, env={}, timeout=20)
    m = re.search(r"(\d+)\.(\d+)\.(\d+)", out.stdout)
    if out.returncode != 0 or not m or tuple(map(int, m.groups())) < (0, 8, 0):
        raise UnsupportedEnvironment(f"bwrap >= 0.8.0 required (--disable-userns): {out.stdout.strip()!r}")
    for kind in NS_KINDS:
        if not os.path.exists(f"/proc/self/ns/{kind}"):
            raise UnsupportedEnvironment(f"kernel lacks /proc/self/ns/{kind}")
    return dict(bwrap=BWRAP, bwrap_sha256=sha256_file(BWRAP), bwrap_version=out.stdout.strip(), uid=os.getuid(),
                kernel=os.uname().release)


def _is_under(path: str, root: str) -> bool:
    return path == root or path.startswith(root.rstrip("/") + "/")


def _symlinks_on_path(path: str) -> list:
    """Every symlink (as (location, link text)) met while resolving ``path`` (components and final file)."""
    out, pending = [], path
    for _ in range(40):
        parts, cur, hop = Path(pending).parts, "/", False
        for i, part in enumerate(parts[1:], 1):
            nxt = os.path.join(cur, part)
            if os.path.islink(nxt):
                text = os.readlink(nxt)
                out.append((nxt, text))
                tgt = text if os.path.isabs(text) else os.path.join(cur, text)
                pending, hop = os.path.normpath(os.path.join(tgt, *parts[i + 1:])), True
                break
            cur = nxt
        if not hop:
            return out
    raise UnsupportedEnvironment(f"symlink chain too deep: {path}")


@functools.lru_cache(maxsize=8)
def _runtime_layout(python: str):
    """Explicit runtime exposure for ``python``: (prefix dirs, [(real_path, dest)], symlink_pairs, root_symlinks)."""
    if not os.path.isabs(python) or not os.path.exists(python):
        raise UnsupportedEnvironment(f"interpreter must be an existing absolute path: {python}")
    out = subprocess.run([python, "-I", "-c", "import sys,json;print(json.dumps([sys.prefix,sys.base_prefix]))"],
                         capture_output=True, text=True, env={}, timeout=60)
    if out.returncode != 0:
        raise UnsupportedEnvironment(f"interpreter probe failed: {out.stderr.strip()}")
    prefix, base = (os.path.realpath(p) for p in json.loads(out.stdout))
    dirs = sorted({prefix, base})
    real_py = os.path.realpath(python)
    if not any(_is_under(real_py, d) for d in dirs) and not _is_under(real_py, "/usr"):
        raise UnsupportedEnvironment(f"interpreter {real_py} outside its prefix dirs {dirs}")
    elfs = {real_py}
    for d in dirs:
        for dp, dn, fn in os.walk(d):
            elfs.update(os.path.realpath(os.path.join(dp, f)) for f in fn if f.endswith(".so") or ".so." in f)
    libs, links, root_links = {}, {}, {}
    for elf in sorted(elfs):
        r = subprocess.run([LDD, elf], capture_output=True, text=True, env={}, timeout=60)
        if r.returncode != 0 and "not a dynamic executable" not in (r.stdout + r.stderr):
            continue   # non-ELF look-alike (e.g. text); a genuinely unresolved lib is caught below
        for line in r.stdout.splitlines():
            if "not found" in line:
                if elf == real_py:
                    raise UnsupportedEnvironment(f"unresolved shared library for the interpreter: {line.strip()}")
                continue    # optional extension (e.g. _tkinter) with a missing host lib: simply not importable in the sandbox
            m = re.search(r"(?:=>\s*)?(/\S+)\s+\(0x", line)
            if not m:
                continue
            p = m.group(1)
            real = os.path.realpath(p)
            if any(_is_under(real, d) for d in dirs):
                continue
            libs[real] = real
            dest = os.path.join(os.path.realpath(os.path.dirname(p)), os.path.basename(p))
            if dest != real:
                links[dest] = real
            top = "/" + p.split("/")[1]
            if os.path.islink(top):
                root_links[top] = os.readlink(top)
    for loc, text in _symlinks_on_path(python):      # e.g. venv bin/python -> uv alias dir -> real interpreter dir
        if not any(_is_under(loc, d) for d in dirs):
            root_links[loc] = text
    return tuple(dirs), tuple(sorted(libs.items())), tuple(sorted(links.items())), tuple(sorted(root_links.items()))


# --------------------------------------------------------------------- spec / plan
@dataclass(frozen=True)
class InputArtifact:
    host_path: str       # approved standalone TRAIN-only artifact (a FILE; its directory is never mounted)
    alias: str           # mounted as /in/<alias>
    sha256: str
    approval_id: str     # identifier of the external approval of this artifact (recorded in the witness; not verified here)


@dataclass(frozen=True)
class IsolationSpec:
    python: str
    code_roots: dict                  # name -> host dir (read-only at /code/<name>)
    code_pins: dict                   # name -> tree sha256 (must cover every code root)
    target: str                       # script path relative to /code, e.g. "harness/probe.py"
    target_args: tuple = ()
    inputs: tuple = ()
    output_dir: str = ""
    protected_paths: tuple = ()       # secret / original-repo / raw-data trees: never bound, contained or exposed
    no_expose_paths: tuple = ()       # directories whose own listing must stay hidden (e.g. project root); children may bind
    bootstrap_root: str = "harness"   # code root holding os_isolation.py + os_isolation_bootstrap.py
    timeout_s: float = 120.0


def _real(path) -> str:
    return os.path.realpath(path)


def _check_not_broad(src: str, label: str, protected, no_expose):
    """``protected``: trees that may be neither bound, contained in a bind, nor exposed by an ancestor bind (secrets,
    original repos, raw data roots). ``no_expose``: directories that must not be bound or have an ancestor-or-equal bound
    (home, an input's parent dir, the project root), but whose explicitly listed descendants may be."""
    if src in BROAD_DENY:
        raise BroadMountError(f"{label}: broad mount source refused: {src}")
    for p in protected:
        if p == src or _is_under(p, src) or _is_under(src, p):
            raise BroadMountError(f"{label}: {src} overlaps protected path {p} (ancestor/descendant/equal)")
    for p in no_expose:
        if p == src or _is_under(p, src):
            raise BroadMountError(f"{label}: {src} would expose {p}")


@dataclass(frozen=True)
class Plan:
    spec: IsolationSpec
    runtime_dirs: tuple
    runtime_libs: tuple
    mounts: tuple          # ((kind, src, dest)) for audit/receipt: kind in ro-bind/bind/tmpfs/proc/dev/symlink
    protected: tuple
    no_expose: tuple


def _protected_sets(spec: IsolationSpec):
    prot = sorted({_real(p) for p in spec.protected_paths})
    exp = {_real(str(Path.home())), "/home", "/root", *(_real(p) for p in spec.no_expose_paths)}
    for a in spec.inputs:
        exp.add(_real(os.path.dirname(a.host_path)))        # an input's parent directory is never mountable
    return prot, sorted(exp)


def build_plan(spec: IsolationSpec, *, host_precheck: bool = True) -> Plan:
    check_environment()
    if not os.path.isabs(spec.python):
        raise OSIsolationError("python must be absolute")
    protected, no_expose = _protected_sets(spec)
    dirs, libs, links, root_links = _runtime_layout(spec.python)
    mounts = []
    for d in dirs:
        _check_not_broad(d, "runtime dir", protected, no_expose)
        mounts.append(("ro-bind", d, d))
    for real, dest in libs:
        mounts.append(("ro-bind", real, real))
    for dest, real in links:
        mounts.append(("symlink", real, dest))
    for top, target in root_links:
        mounts.append(("symlink", target, top))
    if set(spec.code_roots) != set(spec.code_pins):
        raise PinMismatch("code_pins must cover exactly the code roots")
    if spec.bootstrap_root not in spec.code_roots:
        raise OSIsolationError("bootstrap_root must be one of the code roots")
    for name, host in sorted(spec.code_roots.items()):
        if not re.fullmatch(r"[A-Za-z0-9_.-]+", name):
            raise OSIsolationError(f"bad code root name {name!r}")
        src = _real(host)
        if src != str(host) or not os.path.isdir(src) or os.path.islink(host):
            raise OSIsolationError(f"code root must be a real, already-resolved directory: {host}")
        _check_not_broad(src, f"code root {name}", protected, no_expose)
        if host_precheck and tree_sha256(src) != spec.code_pins[name]:
            raise PinMismatch(f"code root {name}: tree sha256 differs from pin")
        mounts.append(("ro-bind", src, f"{CODE_MOUNT}/{name}"))
    for need in ("os_isolation.py", BOOTSTRAP_NAME):
        if not os.path.isfile(os.path.join(spec.code_roots[spec.bootstrap_root], need)):
            raise OSIsolationError(f"bootstrap root lacks {need}")
    if not spec.target.startswith(tuple(f"{n}/" for n in spec.code_roots)):
        raise OSIsolationError("target must live under a pinned code root")
    aliases = set()
    for a in spec.inputs:
        if not re.fullmatch(r"[A-Za-z0-9_.-]+", a.alias) or a.alias in aliases:
            raise OSIsolationError(f"bad/duplicate input alias {a.alias!r}")
        aliases.add(a.alias)
        if not a.approval_id or not isinstance(a.approval_id, str):
            raise OSIsolationError(f"{a.alias}: approval_id missing")
        if not os.path.isabs(a.host_path) or os.path.islink(a.host_path) or not os.path.isfile(a.host_path):
            raise OSIsolationError(f"{a.alias}: input must be an absolute plain file (no symlink)")
        src = _real(a.host_path)
        if src != a.host_path:
            raise OSIsolationError(f"{a.alias}: input path must be already resolved")
        for p in protected:
            if p == src or _is_under(src, p):
                raise BroadMountError(f"{a.alias}: input lies inside protected path {p}")
        if not re.fullmatch(r"[0-9a-f]{64}", a.sha256):
            raise PinMismatch(f"{a.alias}: malformed sha256 pin")
        if host_precheck and sha256_file(src) != a.sha256:
            raise PinMismatch(f"{a.alias}: input sha256 differs from pin")
        mounts.append(("ro-bind", src, f"{IN_MOUNT}/{a.alias}"))
    out = spec.output_dir
    if not out or not os.path.isabs(out) or os.path.islink(out) or not os.path.isdir(out) or _real(out) != out:
        raise OSIsolationError("output_dir must be an existing, resolved, non-symlink absolute directory")
    _check_not_broad(out, "output dir", protected, no_expose)
    for name, host in spec.code_roots.items():
        if _is_under(out, host) or _is_under(host, out):
            raise BroadMountError("output dir overlaps a code root (source writes must stay impossible)")
    for a in spec.inputs:
        if _is_under(a.host_path, out):
            raise BroadMountError("output dir contains an input artifact")
    mounts.append(("bind", out, OUT_MOUNT))
    return Plan(spec=spec, runtime_dirs=dirs, runtime_libs=tuple(r for r, _ in libs), mounts=tuple(mounts),
                protected=tuple(protected), no_expose=tuple(no_expose))


def build_argv(plan: Plan, info_fd: int) -> list:
    a = [BWRAP, "--unshare-user", "--unshare-pid", "--unshare-net", "--unshare-ipc", "--unshare-uts", "--unshare-cgroup",
         "--disable-userns", "--die-with-parent", "--new-session", "--clearenv", "--cap-drop", "ALL",
         "--hostname", "r5-isolated", "--info-fd", str(info_fd),
         "--proc", "/proc", "--dev", "/dev", "--tmpfs", "/tmp", "--size", str(SCRATCH_BYTES), "--tmpfs", SCRATCH_MOUNT]
    for kind, src, dest in plan.mounts:
        a += [f"--{kind}", src, dest] if kind != "symlink" else ["--symlink", src, dest]
    a += ["--remount-ro", "/",
          "--setenv", "HOME", SCRATCH_MOUNT, "--setenv", "TMPDIR", "/tmp", "--setenv", "LC_ALL", "C.UTF-8",
          "--setenv", "PYTHONDONTWRITEBYTECODE", "1", "--setenv", "PYTHONNOUSERSITE", "1",
          "--chdir", SCRATCH_MOUNT, "--",
          plan.spec.python, "-I", "-B", f"{CODE_MOUNT}/{plan.spec.bootstrap_root}/{BOOTSTRAP_NAME}"]
    return a


# bwrap options accepted by the auditor (name -> number of arguments); anything else is refused (fail closed)
_ARITY = {"--unshare-user": 0, "--unshare-pid": 0, "--unshare-net": 0, "--unshare-ipc": 0, "--unshare-uts": 0,
          "--unshare-cgroup": 0, "--disable-userns": 0, "--die-with-parent": 0, "--new-session": 0, "--clearenv": 0,
          "--cap-drop": 1, "--hostname": 1, "--info-fd": 1, "--proc": 1, "--dev": 1, "--tmpfs": 1, "--size": 1,
          "--ro-bind": 2, "--bind": 2, "--symlink": 2, "--remount-ro": 1, "--setenv": 2, "--chdir": 1}
_REQUIRED_FLAGS = ("--unshare-user", "--unshare-pid", "--unshare-net", "--unshare-ipc", "--unshare-uts", "--unshare-cgroup",
                   "--disable-userns", "--die-with-parent", "--new-session", "--clearenv")


def audit_argv(argv, protected=(), no_expose=()) -> dict:
    """Independent parse of a bwrap argv. Refuses broad/unknown/unsafe mounts and missing isolation flags.
    Returns {'binds': [(kind, src, dest)], 'mount_points': set, 'command': [...]}."""
    if not argv or argv[0] != BWRAP:
        raise BroadMountError("argv[0] must be /usr/bin/bwrap")
    prot = [_real(p) for p in protected]
    seen, binds, mount_points, i = [], [], {"/"}, 1
    while i < len(argv):
        tok = argv[i]
        if tok == "--":
            i += 1
            break
        if tok not in _ARITY:
            raise BroadMountError(f"unknown or forbidden bwrap option: {tok}")
        n = _ARITY[tok]
        args = argv[i + 1:i + 1 + n]
        if len(args) != n:
            raise BroadMountError(f"{tok}: missing arguments")
        seen.append(tok)
        if tok in ("--ro-bind", "--bind"):
            src, dest = args
            if not os.path.isabs(src) or ".." in Path(src).parts or not os.path.isabs(dest):
                raise BroadMountError(f"relative bind path: {args}")
            if dest == "/" or dest in ("/proc", "/dev", "/tmp", SCRATCH_MOUNT):
                raise BroadMountError(f"bind over reserved destination {dest}")
            _check_not_broad(_real(src), tok, prot, [_real(p) for p in no_expose])
            if tok == "--bind" and dest != OUT_MOUNT:
                raise BroadMountError("only the output directory may be writable")
            binds.append((tok[2:], src, dest))
            mount_points.add(dest)
        elif tok in ("--tmpfs", "--proc", "--dev"):
            mount_points.add(args[0])
            if tok == "--dev":
                mount_points |= DEV_MOUNTS
        elif tok == "--cap-drop" and args != ["ALL"]:
            raise BroadMountError("only --cap-drop ALL is accepted")
        i += 1 + n
    for flag in _REQUIRED_FLAGS:
        if flag not in seen:
            raise BroadMountError(f"required isolation flag missing: {flag}")
    if "--share-net" in argv:
        raise BroadMountError("network must not be shared")
    if not any(argv[j] == "--cap-drop" and argv[j + 1] == "ALL" for j in range(len(argv) - 1)):
        raise BroadMountError("--cap-drop ALL missing")
    return dict(binds=binds, mount_points=mount_points, command=list(argv[i:]))


# --------------------------------------------------------------------- live probing (host and sandbox side)
def _unescape(s: str) -> str:
    return re.sub(r"\\([0-7]{3})", lambda m: chr(int(m.group(1), 8)), s)


def parse_mountinfo(text: str) -> list:
    rows = []
    for line in text.splitlines():
        f = line.split()
        if "-" not in f:
            continue
        sep = f.index("-")
        rows.append(dict(mount_point=_unescape(f[4]), options=f[5].split(","), fstype=f[sep + 1]))
    return rows


def probe_self() -> dict:
    """Live kernel view from inside the process (no trust in any file the launcher supplied)."""
    fds = {}
    for name in os.listdir("/proc/self/fd"):
        try:
            fds[name] = os.readlink(f"/proc/self/fd/{name}")
        except OSError:
            pass   # the listing fd itself
    status = dict(line.split(":\t", 1) for line in Path("/proc/self/status").read_text().splitlines() if ":\t" in line)
    ifaces = [ln.split(":")[0].strip() for ln in Path("/proc/net/dev").read_text().splitlines()[2:]]
    return dict(
        ns={k: os.readlink(f"/proc/self/ns/{k}") for k in NS_KINDS},
        mounts=parse_mountinfo(Path("/proc/self/mountinfo").read_text()),
        root=sorted(os.listdir("/")), fds=fds, cap_eff=status.get("CapEff", "").strip(),
        no_new_privs=status.get("NoNewPrivs", "").strip(), ifaces=ifaces, pid=os.getpid(), uid=os.getuid())


def _read_nsids(pid) -> dict:
    return {k: os.readlink(f"/proc/{pid}/ns/{k}") for k in NS_KINDS}


def verify_probe(probe: dict, exp: dict) -> list:
    """Compare a live probe with the host observation/plan; returns the list of violations (empty = boundary holds)."""
    bad = []
    for k in NS_KINDS:
        if probe["ns"].get(k) != exp["child_ns"].get(k):
            bad.append(f"ns {k}: {probe['ns'].get(k)} != host-observed {exp['child_ns'].get(k)}")
        if probe["ns"].get(k) == exp["host_ns"].get(k):
            bad.append(f"ns {k}: identical to the launcher's namespace (not isolated)")
    points = {m["mount_point"]: m for m in probe["mounts"]}
    extra = sorted(set(points) - set(exp["mount_points"]))
    if extra:
        bad.append(f"unexpected mounts: {extra[:8]}")
    missing = sorted(set(exp["mount_points"]) - set(points) - DEV_MOUNTS)
    if missing:
        bad.append(f"expected mounts absent: {missing[:8]}")
    for dest in exp["ro_mounts"]:
        if dest in points and "ro" not in points[dest]["options"]:
            bad.append(f"{dest} is writable but planned read-only")
    if probe["root"] != sorted(exp["root"]):
        bad.append(f"root listing {probe['root']} != planned {sorted(exp['root'])}")
    for fd, target in probe["fds"].items():
        if int(fd) > 2 and target != f"/proc/{probe['pid']}/fd":     # the directory fd used for this very listing
            bad.append(f"inherited fd {fd} -> {target}")
        if int(fd) <= 2 and not (target.startswith(("pipe:", "socket:")) or target.startswith("/dev/")):
            bad.append(f"stdio fd {fd} -> {target}")
    if probe["cap_eff"].strip("0") != "":
        bad.append(f"effective capabilities not empty: {probe['cap_eff']}")
    if probe["no_new_privs"] != "1":
        bad.append("no_new_privs not set")
    if [i for i in probe["ifaces"] if i != "lo"]:
        bad.append(f"network interfaces present: {probe['ifaces']}")
    if probe["pid"] > 16:
        bad.append(f"pid {probe['pid']} suggests the host pid namespace")
    return bad


# Sandbox-side state (set only by the bootstrap after it verified the live boundary and the pins).
_ACTIVE: dict | None = None


def activate(expectation: dict, witness: dict):
    global _ACTIVE
    _ACTIVE = dict(expectation=expectation, witness=witness)


def active_witness() -> dict:
    """Re-probe the live sandbox against the host observation recorded at launch. Refuses outside a verified launch."""
    if _ACTIVE is None:
        raise WitnessError("process was not started by the verified bwrap launcher bootstrap (no OS-level boundary witness)")
    bad = verify_probe(probe_self(), _ACTIVE["expectation"])
    if bad:
        raise WitnessError("live OS boundary no longer matches the host observation: " + "; ".join(bad))
    return dict(_ACTIVE["witness"], reverified_live=True)


def require_real_run_release(witness: dict):
    """The release gate. Closed: no release acceptance record exists, so even a verified boundary does not open real data."""
    if REAL_RUN_RELEASE_ACCEPTANCE_SHA256 is None:
        raise WitnessError("real-data release closed: no Director release acceptance record / trusted TRAIN-only input "
                           "artifacts exist; required: " + " | ".join(RELEASE_REQUIREMENTS))
    raise WitnessError("release acceptance verification is not implemented (fails closed)")

@dataclass
class LaunchResult:
    status: str                      # OK | TARGET_FAILED | TARGET_SIGNALED | BOOTSTRAP_FAILED | SANDBOX_SETUP_FAILED
    #                                # | HOST_VERIFICATION_FAILED | TIMEOUT
    returncode: int | None
    stdout: str
    stderr: str
    detail: str = ""
    witness: dict | None = None
    argv: list = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return self.status == "OK"


class _Pump(threading.Thread):
    def __init__(self, stream, ready_marker=None):
        super().__init__(daemon=True)
        self.stream, self.buf, self.ready = stream, [], threading.Event()
        self.marker = ready_marker
        self.start()

    def run(self):
        for raw in iter(self.stream.readline, b""):
            line = raw.decode("utf-8", "replace")
            self.buf.append(line)
            if self.marker and line.startswith(self.marker):
                self.ready.set()
        self.ready.set()

    def text(self):
        return "".join(self.buf)


def _marker_json(text: str, marker: str):
    for line in text.splitlines():
        if line.startswith(marker):
            return json.loads(line[len(marker):].strip().rstrip("@"))
    return None


def launch(spec: IsolationSpec, *, host_precheck: bool = True, _tamper=None, _argv_hook=None,
           _prefix=()) -> LaunchResult:
    """Launch the spec's target inside the bwrap boundary and return a LaunchResult.

    Pre-launch spec violations raise OSIsolationError subclasses; everything after the process starts is reported as a
    status. ``_tamper``/``_argv_hook`` exist only so tests can forge the handshake / feed a broad argv to the auditor;
    ``_prefix`` lets tests run the audited bwrap argv under ``strace`` as an independent syscall-level witness."""
    plan = build_plan(spec, host_precheck=host_precheck)
    info_r, info_w = os.pipe()
    argv = build_argv(plan, info_w)
    if _argv_hook:
        argv = _argv_hook(argv)
    audit = audit_argv(argv, plan.protected, plan.no_expose)
    nonce = secrets.token_hex(16)
    proc = subprocess.Popen([*_prefix, *argv], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, env={},
                            close_fds=True, pass_fds=(info_w,), start_new_session=True, cwd="/")
    os.close(info_w)
    out_pump, err_pump = _Pump(proc.stdout), _Pump(proc.stderr, MARK_READY)

    def finish(status, detail="", witness=None):
        try:
            proc.stdin.close()
        except OSError:
            pass
        if proc.poll() is None:
            try:
                os.killpg(proc.pid, signal.SIGKILL)
            except OSError:
                pass
        proc.wait()
        out_pump.join(5), err_pump.join(5)
        rc = proc.returncode
        return LaunchResult(status, rc, out_pump.text(), err_pump.text(), detail, witness, argv)

    deadline = time.monotonic() + spec.timeout_s
    info = _read_info(info_r, deadline)
    if "child-pid" not in info:
        rc = proc.wait(timeout=max(1, deadline - time.monotonic()))
        out_pump.join(5), err_pump.join(5)
        return LaunchResult("SANDBOX_SETUP_FAILED", rc, out_pump.text(), err_pump.text(),
                            "bwrap did not report a sandbox (unsupported environment?)", None, argv)
    if not err_pump.ready.wait(max(0.1, deadline - time.monotonic())):
        return finish("TIMEOUT", "bootstrap never became ready")
    ready = _marker_json(err_pump.text(), MARK_READY)
    if ready is None:
        return _classify_end(proc, out_pump, err_pump, argv, deadline, finish, nonce, None)
    host_ns = _read_nsids("self")
    child_ns = {k: ready["ns"][k] for k in NS_KINDS}
    problems = []
    for k, key in INFO_NS_KEYS.items():       # bwrap's own report of the namespaces it created
        want = f"{k}:[{info.get(key)}]"
        if info.get(key) is None or want != child_ns[k]:
            problems.append(f"{k}: bwrap info {want} != child {child_ns[k]}")
    if [k for k in NS_KINDS if child_ns[k] == host_ns[k]]:
        problems.append("namespace shared with launcher: " + ",".join(k for k in NS_KINDS if child_ns[k] == host_ns[k]))
    pid = info["child-pid"]
    proc_view = {}
    try:        # independent host-side view of the live sandbox (when /proc/<pid> is readable to us)
        proc_view = dict(ns=_read_nsids(pid), mounts=parse_mountinfo(Path(f"/proc/{pid}/mountinfo").read_text()))
        if proc_view["ns"] != child_ns:
            problems.append("/proc/<pid>/ns differs from the child's self-report")
        hm = {m["mount_point"] for m in proc_view["mounts"]}
        if hm - audit["mount_points"]:
            problems.append(f"host-observed unexpected mounts: {sorted(hm - audit['mount_points'])[:8]}")
    except OSError:
        proc_view = {}
    top = sorted({d.split("/")[1] for d in audit["mount_points"] if d != "/"} | {m[2].split("/")[1] for m in plan.mounts})
    exp = dict(child_ns=child_ns, host_ns=host_ns, mount_points=sorted(audit["mount_points"]), root=top,
               ro_mounts=sorted(d for k, _, d in audit["binds"] if k == "ro-bind"))
    if problems:
        return finish("HOST_VERIFICATION_FAILED", "; ".join(problems))
    witness = dict(
        schema=WITNESS_SCHEMA, profile=PROFILE, nonce=nonce, launcher_pid=os.getpid(), bwrap_child_pid=pid,
        argv_sha256=hashlib.sha256(json.dumps(argv).encode()).hexdigest(), host_ns=host_ns, child_ns=child_ns,
        bwrap_info=info, host_proc_view_available=bool(proc_view), mount_points=exp["mount_points"],
        root_listing=top, code_pins=dict(spec.code_pins),
        inputs={a.alias: dict(sha256=a.sha256, approval_id=a.approval_id) for a in spec.inputs},
        runtime_dirs=list(plan.runtime_dirs), runtime_lib_files=list(plan.runtime_libs), network="unshared (no interfaces)",
        bootstrap_exe=f"{CODE_MOUNT}/{spec.bootstrap_root}/{BOOTSTRAP_NAME}")
    hand = dict(nonce=nonce, expectation=exp, witness=witness, code_pins=spec.code_pins,
                inputs={a.alias: a.sha256 for a in spec.inputs}, target=f"{CODE_MOUNT}/{spec.target}",
                target_args=list(spec.target_args))
    if _tamper:
        hand = _tamper(hand)
    try:
        if hand is not None:        # (tests: None = the launcher sends no handshake at all)
            proc.stdin.write((json.dumps(hand) + "\n").encode())
            proc.stdin.flush()
        proc.stdin.close()
    except OSError:
        pass
    return _classify_end(proc, out_pump, err_pump, argv, deadline, finish, nonce, witness)


def _read_info(fd: int, deadline: float) -> dict:
    """bwrap's --info-fd JSON. Parse as soon as it is complete (a tracer such as strace may keep the pipe open, so EOF
    is not required); EOF without a document means bwrap failed before creating the sandbox."""
    import select
    raw = b""
    try:
        while time.monotonic() < deadline:
            if not select.select([fd], [], [], 0.2)[0]:
                continue
            chunk = os.read(fd, 65536)
            if not chunk:
                break
            raw += chunk
            try:
                return json.loads(raw.decode())
            except ValueError:
                continue
    finally:
        os.close(fd)
    return {}


def _classify_end(proc, out_pump, err_pump, argv, deadline, finish, nonce, witness):
    try:
        rc = proc.wait(timeout=max(0.1, deadline - time.monotonic()))
    except subprocess.TimeoutExpired:
        return finish("TIMEOUT", "deadline exceeded", witness)
    out_pump.join(5), err_pump.join(5)
    out, err = out_pump.text(), err_pump.text()
    fail = _marker_json(err, MARK_FAIL)
    started = any(ln.startswith(MARK_START) for ln in err.splitlines())
    ended = _marker_json(err, MARK_END)
    mk = lambda status, detail="": LaunchResult(status, rc, out, err, detail, witness, argv)   # noqa: E731
    if fail is not None:
        return mk("BOOTSTRAP_FAILED", json.dumps(fail))
    if not started:
        return mk("BOOTSTRAP_FAILED" if witness else "SANDBOX_SETUP_FAILED", "bootstrap did not reach the target")
    if ended is None:
        sig = rc - 128 if rc and rc > 128 else None
        return mk("TARGET_SIGNALED", f"target ended without completing (signal {sig})" if sig else "target ended abnormally")
    if ended.get("rc") != 0:
        return mk("TARGET_FAILED", f"target exit code {ended.get('rc')}")
    return mk("OK" if rc == 0 else "TARGET_FAILED", "" if rc == 0 else f"bwrap exit {rc}")


def trusted_witness(result: LaunchResult) -> dict:
    """Host-side acceptance of a launch as an OS-boundary witness: only a launched, host-verified, bootstrap-passed run."""
    if result.witness is None or result.status not in ("OK", "TARGET_FAILED"):
        raise WitnessError(f"no OS-boundary witness: launch status {result.status}")
    return result.witness
